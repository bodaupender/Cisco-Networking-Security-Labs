! ============================================
! HR-SW — Access Switch, VLAN 10 (HR)
! Uplink: Gig1/0/1 -> CORE-SW
! Mgmt: 192.16899.11 | GW: 192.168.99.1
! Model: 3650 (ports = gig1/0/x)
! ============================================

enable
configure terminal
hostname SALES-SW
no ip domain-lookup
service password-encryption
enable secret <REDACTED>
banner motd #AUTHORIZED ACCESS ONLY#
line console 0
 password <REDACTED>
 login
 exec-timeout 5 0
 logging synchronous
exit
username admin privilege 15 secret <REDACTED>
ip domain-name company.local
crypto key generate rsa modulus 1024
line vty 0 4
 transport input ssh
 login local
 exec-timeout 5 0
exit

! ===== VLANs =====
vlan 40
 name SALES
vlan 99
 name MANAGEMENT
vlan 999
 name UNUSED_BLACKHOLE
exit

! ===== Trunk to CORE-SW =====
interface gig1/0/1
 description UPLINK-TO-CORE-SW
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,99
exit

! ===== Access ports (users) =====
interface range gig1/0/2-9
 description SALES-USERS
 switchport mode access
 switchport access vlan 40
exit

! ===== Management =====
interface vlan 99
 ip address 192.168.99.14  255.255.255.0
 no shutdown
exit
ip default-gateway 192.168.99.1

! ===== DHCP Snooping =====
ip dhcp snooping
ip dhcp snooping vlan 10,20,30,40,50
no ip dhcp snooping information option
interface gig1/0/1
 ip dhcp snooping trust
exit

! ===== Dynamic ARP Inspection =====
ip arp inspection vlan 10,20,30,40,50
interface gig1/0/1
 ip arp inspection trust
exit

! ===== STP Security =====
spanning-tree mode rapid-pvst
interface range gig1/0/2-9
 spanning-tree portfast
 spanning-tree bpduguard enable
exit
interface gig1/0/1
 spanning-tree guard root
exit

! ===== Port Security =====
interface range gig1/0/2-9
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
exit

configure terminal
interface gig1/0/9
 switchport port-security maximum 4
 shutdown
 no shutdown
exit
end
copy running-config startup-config


! ===== Syslog =====
logging host 192.168.50.12

! ===== Unused Ports (hardened) =====
interface range gig1/0/10-24 , gig1/1/1-4
 description UNUSED-HARDENED
 shutdown
 switchport mode access
 switchport access vlan 999
exit

end
copy running-config startup-config
