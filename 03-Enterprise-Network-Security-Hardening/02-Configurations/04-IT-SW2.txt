1)ERASE ALL DATA BEFORE CONFIGURED

enable
write erase
delete flash:vlan.dat
reload
exit

2)ASSIGN NAME

enable
configure terminal
hostname IT-SW2
end


3)VLAN CONFIGURATION

enable
configure terminal

vlan 30
 name IT
exit

vlan 99
 name MANAGEMENT
exit

vlan 999
 name NATIVE-MGMT
exit

end

3.1)VERIFY

show vlan brief
show ip interface brief
show running-config | section interface Vlan


4)PORT ASSIGNMENT

enable
configure terminal

interface range fa0/2 - 7
 description HR-Users
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
exit
end
write memory

4.1)VERIFY

show vlan brief

5)TRUNKING

enable
configure terminal

interface fa0/1
 description TRUNK-TO-IT-SW1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99,999
 no shutdown
exit

end
write memory

5.1)VERIFY
show cdp neighbors
show interfaces trunk
