end

6)DHCP

6.1)DHCP Address Reservation  
[.1  = VLAN gateway
.2-.20 = reserved/static infrastructure
.21-.254 = DHCP client range]

enable
configure terminal

ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.30.1 10.10.30.20
ip dhcp excluded-address 10.10.50.1 10.10.50.20
ip dhcp excluded-address 10.10.60.1 10.10.60.20

end


6.2)VERIFY

show ip dhcp excluded-address

6.3)Create DHCP pool for HR

enable
configure terminal

ip dhcp pool VLAN10-HR
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 8.8.8.8
exit
